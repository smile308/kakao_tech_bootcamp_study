## 13.2 `src/api/api.js`: 모든 HTTP 요청의 공통 입구

### 파일을 지금 읽는 이유

Login·Signup page는 직접 `fetch`를 호출하지 않습니다. 이 파일이 API base URL, JSON header,
Cookie credentials, access token header, 401 refresh, 오류 변환, 재요청을 모두 공통 처리합니다.
이 파일을 먼저 읽어야 `authApi.login()`이 실제로 무엇을 반환하는지 이해할 수 있습니다.

### 파일 책임과 연결

- 실제 파일: `/Users/miles/Documents/GitHub/KTB4_Miles_Week12_Front/src/api/api.js`
- 호출하는 파일: `authApi.js`, `userApi.js`, 이후 `postApi.js` 등 모든 API module
- 호출되는 파일: 12번에서 확인한 `authStorage.js`, 브라우저 `fetch`, `window.location`
- 외부 입력: endpoint, fetch options, localStorage token, HTTP response
- 생성·변경 값: request headers, shared `refreshPromise`, access token 저장·삭제
- 반환값 사용 위치: API module과 page의 `await` 결과
- 예외 지점: fetch rejection, JSON parse 실패, non-2xx response, refresh 실패
- 다음 파일: `authApi.js`, `userApi.js`

### 전체 실제 코드

```javascript
import { authStorage } from "../auth/authStorage.js";

const API_BASE_URL = import.meta.env.VITE_API_BASE_URL ?? "http://localhost:8080";
const PUBLIC_PATHS = ["/login", "/signup"];
const NO_REFRESH_REQUESTS = new Set([
    "POST /sessions",
    "POST /sessions/refresh",
    "DELETE /sessions",
    "POST /users",
]);

let refreshPromise = null;

function isRefreshExcludedRequest(endpoint, options = {}) {
    const method = (options.method ?? "GET").toUpperCase();
    return NO_REFRESH_REQUESTS.has(`${method} ${endpoint}`);
}

async function sendRequest(endpoint, options = {}, includeAccessToken = true) {
    const accessToken = authStorage.getAccessToken();
    const headers = {
        "Content-Type": "application/json",
        ...options.headers,
    };

    if (includeAccessToken && accessToken) {
        headers.Authorization = `Bearer ${accessToken}`;
    }

    const response = await fetch(`${API_BASE_URL}${endpoint}`, {
        ...options,
        headers,
        credentials: "include",
    });

    return {
        response,
        data: await readResponseData(response),
    };
}

async function readResponseData(response) {
    const contentType = response.headers.get("content-type");
    const responseText = await response.text();

    if (!responseText) {
        return null;
    }

    return contentType?.includes("application/json")
        ? JSON.parse(responseText)
        : responseText;
}

function createRequestError(response, data) {
    const errorCode =
        typeof data === "string"
            ? data
            : data?.code ?? data?.message ?? "UNKNOWN_ERROR";
    const errorMessage =
        typeof data === "string"
            ? data
            : data?.message ?? "요청 처리에 실패했습니다.";

    const error = new Error(errorMessage);
    error.code = errorCode;
    error.status = response.status;
    error.data = data;
    return error;
}

function clearAuthentication() {
    authStorage.removeAccessToken();

    if (!PUBLIC_PATHS.includes(window.location.pathname)) {
        window.location.replace("/login");
    }
}

async function performRefresh() {
    const result = await sendRequest(
        "/sessions/refresh",
        { method: "POST" },
        false,
    );

    if (!result.response.ok) {
        throw createRequestError(result.response, result.data);
    }

    if (!result.data?.accessToken) {
        throw new Error("액세스 토큰 재발급 응답이 올바르지 않습니다.");
    }

    authStorage.setAccessToken(result.data.accessToken);
    return result.data;
}

export function refreshAccessToken() {
    if (!refreshPromise) {
        refreshPromise = performRefresh().finally(() => {
            refreshPromise = null;
        });
    }

    return refreshPromise;
}

export async function request(endpoint, options = {}) {
    let result = await sendRequest(endpoint, options);

    if (
        result.response.status === 401 &&
        !isRefreshExcludedRequest(endpoint, options)
    ) {
        try {
            await refreshAccessToken();
            result = await sendRequest(endpoint, options);
        } catch (error) {
            if (error.status === 401) {
                clearAuthentication();
            }

            throw error;
        }
    }

    if (!result.response.ok) {
        if (result.response.status === 401) {
            clearAuthentication();
        }

        throw createRequestError(result.response, result.data);
    }

    return result.data;
}
```

### 코드 일부와 바로 아래 설명: 환경변수와 refresh 제외 목록

```javascript
const API_BASE_URL = import.meta.env.VITE_API_BASE_URL ?? "http://localhost:8080"; // Vite 환경변수가 없으면 local backend를 사용한다.
const PUBLIC_PATHS = ["/login", "/signup"]; // 인증 실패 redirect 후 다시 replace하지 않을 공개 경로다.
const NO_REFRESH_REQUESTS = new Set([ // 401이어도 공통 refresh를 다시 시도하지 않을 요청 조합을 저장한다.
    "POST /sessions", // 로그인 요청 자체가 실패한 경우다.
    "POST /sessions/refresh", // refresh 요청이 다시 refresh를 부르는 무한 loop를 막는다.
    "DELETE /sessions", // 로그아웃 요청에서 추가 refresh를 하지 않는다.
    "POST /users", // 회원가입은 로그인 전 공개 요청이다.
]);

let refreshPromise = null; // 동시에 진행 중인 refresh Promise 하나를 공유한다.
```

`import.meta.env`는 Vite가 build 시 주입하는 환경변수 객체입니다. `VITE_` 접두사가 있는
변수만 client source에 노출될 수 있습니다. `??`는 값이 `null` 또는 `undefined`일 때만
local fallback을 사용합니다.

`Set`은 중복 없는 값 collection이고 `.has()`로 포함 여부를 확인합니다. 여기서는 method와
endpoint를 합친 문자열을 저장해 refresh 제외 여부를 빠르게 검사합니다.

`refreshPromise`는 사용자 한 명의 token만 저장하는 변수가 아닙니다. 여러 요청이 동시에
401을 받았을 때 첫 refresh의 Promise를 공유해 refresh HTTP 요청을 하나로 합치는 동시성
제어 변수입니다. `null`이면 진행 중인 refresh가 없다는 뜻입니다.

### 코드 일부와 바로 아래 설명: 공통 request 생성

```javascript
function isRefreshExcludedRequest(endpoint, options = {}) {
    const method = (options.method ?? "GET").toUpperCase(); // method가 없으면 GET으로 간주하고 대문자로 통일한다.
    return NO_REFRESH_REQUESTS.has(`${method} ${endpoint}`); // 예: "POST /sessions" 조합을 Set에서 찾는다.
}

async function sendRequest(endpoint, options = {}, includeAccessToken = true) {
    const accessToken = authStorage.getAccessToken(); // 12번에서 확인한 localStorage wrapper로 token을 읽는다.
    const headers = {
        "Content-Type": "application/json", // 기본 요청 body가 JSON임을 표시한다.
        ...options.headers, // 호출자가 전달한 header로 기본값을 덮어쓸 수 있다.
    };

    if (includeAccessToken && accessToken) {
        headers.Authorization = `Bearer ${accessToken}`; // 보호 요청에 Authorization header를 추가한다.
    }
```

`options = {}`는 호출자가 두 번째 인자를 생략했을 때 빈 객체를 사용하게 하는 default
parameter입니다. `...options.headers`는 object spread이며, 호출자 header를 기존 headers에
합칩니다. 뒤에 펼친 값이 같은 key를 가지면 앞의 값을 덮어씁니다.

`includeAccessToken`은 refresh 요청처럼 Cookie만 사용하고 access token header를 넣으면
안 되는 경우를 위한 flag입니다. 일반 `request()`는 기본값 `true`로 호출되고,
`performRefresh()`만 `false`로 호출합니다.

```javascript
    const response = await fetch(`${API_BASE_URL}${endpoint}`, { // base URL과 endpoint를 합쳐 브라우저 fetch를 실행한다.
        ...options, // method, body, signal 등 호출자가 준 fetch 옵션을 전달한다.
        headers, // 위에서 완성한 JSON·Authorization header를 사용한다.
        credentials: "include", // refresh Cookie 같은 cross-origin credential을 함께 보낸다.
    });

    return {
        response, // status·ok·headers를 후속 로직에서 사용한다.
        data: await readResponseData(response), // response body를 읽어 data로 함께 반환한다.
    };
}
```

`fetch`는 HTTP 오류 status가 400·500이어도 Promise를 reject하지 않고 Response를 resolve합니다.
따라서 이 함수는 `response`를 반환하고, 나중에 `request()`가 `response.ok`를 검사합니다.
`credentials: "include"`는 브라우저가 Cookie를 요청에 포함하고 Set-Cookie 응답을 처리할
수 있게 하는 옵션입니다. Cookie 정책 자체는 백엔드와 브라우저의 CORS 설정이 함께 결정합니다.

### 코드 일부와 바로 아래 설명: response body 읽기와 오류 객체 생성

```javascript
async function readResponseData(response) {
    const contentType = response.headers.get("content-type"); // 응답 Content-Type header를 읽는다.
    const responseText = await response.text(); // body를 먼저 문자열로 한 번 읽는다.

    if (!responseText) {
        return null; // 204처럼 body가 없으면 null을 반환한다.
    }

    return contentType?.includes("application/json") // Content-Type이 JSON인지 확인한다.
        ? JSON.parse(responseText) // JSON이면 JavaScript object로 파싱한다.
        : responseText; // 아니면 문자열 그대로 반환한다.
}
```

Response body stream은 한 번 읽으면 다시 같은 방식으로 읽을 수 없으므로 `text()`로 먼저
읽고, Content-Type에 따라 `JSON.parse`를 선택합니다. `?.` optional chaining은
`contentType`이 null이어도 `includes` 호출 오류를 내지 않습니다. JSON이라고 표시됐지만
문법이 깨져 있으면 `JSON.parse`가 예외를 던지고, 그 예외는 request 호출자에게 전파됩니다.

```javascript
function createRequestError(response, data) {
    const errorCode = typeof data === "string" // 문자열 body인지 object body인지 분기한다.
        ? data // 문자열이면 그 값을 code로 사용한다.
        : data?.code ?? data?.message ?? "UNKNOWN_ERROR"; // object면 code, message, fallback 순으로 찾는다.
    const errorMessage = typeof data === "string"
        ? data
        : data?.message ?? "요청 처리에 실패했습니다.";

    const error = new Error(errorMessage); // JavaScript Error instance를 만든다.
    error.code = errorCode; // 화면이 오류 종류를 식별할 custom property를 추가한다.
    error.status = response.status; // HTTP status를 Error에 복사한다.
    error.data = data; // 원래 body도 후속 화면·로그에서 사용할 수 있게 보관한다.
    return error; // request의 catch로 전달할 하나의 Error 객체를 반환한다.
}
```

`Error`는 JavaScript 내장 class이고, 여기에 `code`, `status`, `data`를 동적으로 추가할 수
있습니다. 이 객체는 `errorMessage.js`의 `getErrorCode()`와 `getErrorMessage()`가 읽고
Login·Signup page의 field error 문구로 변환합니다.

### 코드 일부와 바로 아래 설명: 401 실패와 인증 초기화

```javascript
function clearAuthentication() {
    authStorage.removeAccessToken(); // localStorage의 access token을 제거한다.

    if (!PUBLIC_PATHS.includes(window.location.pathname)) { // 현재 화면이 login/signup이 아닌지 확인한다.
        window.location.replace("/login"); // 공개 화면이 아니면 브라우저를 로그인 경로로 이동시킨다.
```

`includes`는 배열에 현재 pathname이 있는지 boolean으로 반환합니다. `replace`는 현재

이 함수는 401이라고 해서 모든 요청을 즉시 처리하는 것이 아니라, refresh도 실패했거나
최종 response가 401인 경우에만 `request()`가 호출합니다.

### 코드 일부와 바로 아래 설명: refresh 요청·동시성 공유

```javascript
async function performRefresh() {
    const result = await sendRequest(
        "/sessions/refresh", // 백엔드 refresh endpoint다.
        { method: "POST" }, // refresh token은 Cookie로 보내므로 body는 없다.
        false, // Authorization header를 추가하지 않는다.
    );

    if (!result.response.ok) {
        throw createRequestError(result.response, result.data); // refresh 실패를 표준 Error로 전파한다.
    }

    if (!result.data?.accessToken) {
        throw new Error("액세스 토큰 재발급 응답이 올바르지 않습니다."); // 성공 body에도 token이 없으면 실패로 본다.
    }

    authStorage.setAccessToken(result.data.accessToken); // 새 access token을 localStorage에 저장한다.
    return result.data; // refresh를 기다린 모든 호출자에게 새 응답 data를 전달한다.
}

export function refreshAccessToken() {
    if (!refreshPromise) { // 아직 공동 refresh가 없을 때만 실제 요청을 시작한다.
        refreshPromise = performRefresh().finally(() => {
            refreshPromise = null; // 성공·실패 후 다음 401을 위한 빈 상태로 되돌린다.
        });
    }

    return refreshPromise; // 첫 호출과 이후 동시 호출이 같은 Promise를 받는다.
}
```

`performRefresh()`는 실제 refresh HTTP 호출이고, `refreshAccessToken()`은 이를 공유하는
조정자입니다. 세 요청이 거의 동시에 401을 받으면 첫 요청만 `performRefresh()`를 실행하고,
나머지는 이미 저장된 `refreshPromise`를 반환해 같은 결과를 기다립니다. `finally`는
Promise가 성공하든 실패하든 실행되어 오래된 Promise가 다음 요청에 남지 않게 합니다.

### 코드 일부와 바로 아래 설명: request의 401 재시도

```javascript
export async function request(endpoint, options = {}) {
    let result = await sendRequest(endpoint, options); // 먼저 현재 access token으로 원래 요청을 보낸다.

    if (
        result.response.status === 401 && // 인증 부족 응답인지 확인한다.
        !isRefreshExcludedRequest(endpoint, options) // 로그인·refresh·회원가입 등은 재발급하지 않는다.
    ) {
        try {
            await refreshAccessToken(); // 공동 refresh가 끝날 때까지 기다린다.
            result = await sendRequest(endpoint, options); // 새 token으로 원래 요청을 한 번 재시도한다.
        } catch (error) {
            if (error.status === 401) {
                clearAuthentication(); // refresh 자체도 인증 실패면 local 인증을 제거한다.
            }
            throw error; // 호출 page가 오류 문구를 결정하도록 전파한다.
        }
    }

    if (!result.response.ok) { // 재시도 결과까지 포함해 최종 status를 검사한다.
        if (result.response.status === 401) {
            clearAuthentication(); // 최종 보호 요청이 여전히 401이면 로그인으로 보낸다.
        }
        throw createRequestError(result.response, result.data); // non-2xx를 Error로 바꾼다.
    }

    return result.data; // 성공 body만 API module과 page에 반환한다.
}
```

`request()`의 `return`은 현재 `request` 함수만 종료하고, 호출자인 `authApi.login()`이나
page의 `await`에 값을 전달합니다. refresh가 성공하면 원래 endpoint를 한 번 재요청하지만,
refresh 제외 목록에 있는 로그인·회원가입 요청은 401이어도 refresh loop에 들어가지 않습니다.

## 13.3 `src/api/authApi.js`: 인증 endpoint 이름을 모으는 module

### 파일을 지금 읽는 이유

Page가 endpoint 문자열과 JSON 직렬화를 직접 작성하지 않도록 인증 요청을 하나의 객체로
모읍니다. 실제 HTTP 동작은 `api.js`의 `request()`가 하고, 이 파일은 업무 이름과 payload를
endpoint 계약으로 연결합니다.

### 파일 책임과 연결

- 실제 파일: `/Users/miles/Documents/GitHub/KTB4_Miles_Week12_Front/src/api/authApi.js`
- 호출하는 파일: `LoginPage.jsx`, `SignupPage.jsx`, `AppHeader.jsx`
- 호출되는 파일: `request()`, `refreshAccessToken()`
- 외부 입력: login email/password, signup payload
- 생성·변경 값: JSON request body와 API Promise
- 반환값 사용 위치: LoginPage의 access token 저장, SignupPage의 성공 이동, Header logout
- 예외 지점: `request()`가 만든 Error가 호출자로 전파됩니다.
- 다음 파일: `userApi.js`

### 전체 실제 코드

```javascript
import { refreshAccessToken, request } from "./api.js";

export const authApi = {
    login({ email, password }) {
        return request("/sessions", {
            method: "POST",
            body: JSON.stringify({ email, password }),
        });
    },

    logout() {
        return request("/sessions", { method: "DELETE" });
    },

    refresh() {
        return refreshAccessToken();
```

```javascript
        return false;
    }
    const hasValidLength=password.length>=8 && password.length<=20; // 길이 범위를 계산한다.
    const hasUpperCase=/[A-Z]/.test(password); // 대문자 포함 여부
    const hasLowerCase=/[a-z]/.test(password); // 소문자 포함 여부
    const hasNumber=/[0-9]/.test(password); // 숫자 포함 여부
    const hasSpecialCharacter=PASSWORD_REGEX.test(password); // 특수문자 포함 여부
    return hasValidLength && hasUpperCase && hasLowerCase && hasNumber && hasSpecialCharacter;
}
```

`&&`는 앞 조건이 false면 뒤 조건을 더 평가하지 않는 short-circuit 연산입니다. 모든
password 조건을 만족해야 true가 됩니다. SignupPage의 `passwordCheck` 일치 여부는 이
파일이 아니라 SignupPage의 `validateField()`와 `isValid` 계산에서 별도로 검사합니다.

```javascript

    return new Promise((resolve, reject)=>{ // 완료와 실패를 호출자 await/catch로 연결한다.
        if(!file){
            reject(new Error("파일이 없습니다.")); // 입력이 없으면 실패 Promise를 만든다.
            return; // reject 후에도 아래 FileReader를 만들지 않도록 현재 executor를 끝낸다.
        }
        const reader=new FileReader(); // 브라우저가 제공하는 파일 읽기 객체다.
        reader.onload=()=>{
            resolve(reader.result); // 읽기가 끝난 결과(data:image/...;base64,...)를 전달한다.
        };
        reader.onerror=()=>{
            reject(reader.error); // 읽기 실패 원인을 catch로 보낸다.
        };
        reader.readAsDataURL(file); // 파일을 Data URL 방식으로 읽기 시작한다.
    });
}
```

`FileReader`는 즉시 문자열을 반환하지 않고 이벤트(`onload`, `onerror`)를 나중에 발생시킵니다.
`Promise`로 감싸면 SignupPage에서 `await fileToDataUrl(file)`처럼 순차적인 코드로 기다릴

## 13.8 공통 입력 컴포넌트

### 읽는 이유와 공통 구조

LoginForm과 SignupForm은 같은 입력 UI를 반복해서 만들지 않고 이 컴포넌트들에 props를
전달합니다. 이 파일들은 API를 호출하지 않으며, 부모 page가 가진 state·handler를 DOM
event에 연결하는 표시 계층입니다.

### 묶음 안 파일별 연결 확인

| 파일 | 지금 읽는 이유·책임 | 호출하는/호출되는 코드 | 입력·생성값과 반환 사용 | 예외·다음 흐름 |
|---|---|---|---|---|
| `FormField.jsx` | label·children·field 오류를 한 묶음으로 표시 | LoginForm/SignupForm → children으로 TextInput | `label`, `htmlFor`, `error`를 받아 JSX를 반환 | 자체 예외 없음 → TextInput |
| `TextInput.jsx` | input 공통 class와 모든 HTML props 전달 | LoginForm/SignupForm → 실제 `<input>` | 부모의 `name/value/onChange/onBlur`가 DOM으로 전달 | 자체 예외 없음 → TextArea 패턴 |
| `TextArea.jsx` | 여러 줄 입력의 동일한 props 전달 패턴 확인 | 이후 게시글·댓글 form에서 호출 예정 | `textarea` JSX 반환 | 현재 인증 화면에서는 직접 호출되지 않음 → 게시글 입력 단계 |

| `AppLink.jsx` | route 이동을 공통 wrapper로 사용하는 이유 확인 | LoginForm/SignupForm → React Router `Link` | `to`와 children을 Link로 전달 | Router context 밖이면 framework 오류 가능 → `/login`·`/signup` route |

### 전체 실제 코드

```jsx
import { useEffect } from "react";

function ConfirmModal({
    isOpen,
    title,
    description,
    confirmText = "확인",
    cancelText,
    onConfirm,
    onCancel,
}) {
    useEffect(() => {
        document.body.classList.toggle("modal-open", isOpen);
        return () => document.body.classList.remove("modal-open");
    }, [isOpen]);

    if (!isOpen) {
        return null;
    }

    return (
        <div className="modal-overlay" aria-hidden="false">
            <section className="confirm-modal" role="dialog" aria-modal="true">
                <h2 className="confirm-modal__title">{title}</h2>
                <p className="confirm-modal__description">{description}</p>
                <div className="confirm-modal__actions">
                    {onCancel ? (
                        <button
                            type="button"
                            className="confirm-modal__button confirm-modal__button--cancel"
                            onClick={onCancel}
                        >
                            {cancelText ?? "취소"}
                        </button>
                    ) : null}
                    <button
                        type="button"
                        className="confirm-modal__button confirm-modal__button--confirm"
                        onClick={onConfirm}
                    >
                        {confirmText}
                    </button>
                </div>
            </section>
        </div>
    );
}

export default ConfirmModal;
```

### 코드 일부와 바로 아래 설명

```jsx

    document.body.classList.toggle("modal-open", isOpen); // 열림이면 추가, 닫힘이면 제거한다.
    return () => document.body.classList.remove("modal-open"); // effect가 다시 실행되거나 unmount될 때 정리한다.
}, [isOpen]); // isOpen 변경 때만 effect를 재실행한다.
```

여기서 `useEffect`는 렌더링 결과 자체가 아닌 브라우저 DOM을 변경하는 side effect를
처리합니다. cleanup 함수는 이전 effect가 남긴 전역 class를 제거해 modal이 닫힌 뒤
페이지 스크롤 잠금 같은 스타일이 남지 않게 합니다.

```jsx
if (!isOpen) {
    return null; // 닫힌 상태에서는 modal DOM을 React tree에 만들지 않는다.
}

{onCancel ? ( // 취소 callback이 전달된 경우에만 취소 버튼을 만든다.
    <button type="button" onClick={onCancel}>
        {cancelText ?? "취소"} // cancelText가 null/undefined면 기본 문구를 표시한다.
    </button>
) : null}
```

`return null`은 컴포넌트가 아무 DOM도 렌더링하지 않는 React 규칙입니다. `? :`는 조건부
표현식이고, `??`는 값이 `null` 또는 `undefined`인 경우에만 오른쪽 기본값을 선택합니다.
`onConfirm`은 LoginPage에서 `setIsSuspendedModalOpen(false)`를 호출하므로 확인 클릭의
상태 변경은 부모에서 일어납니다. `ProfileEditPage`와 `PostDetailPage`도 같은 컴포넌트를
사용하지만 각자 다른 `title`, `description`, `onConfirm`, `onCancel`을 전달합니다. 따라서
이 파일은 “정지 계정 전용” 컴포넌트가 아니라 부모가 문구와 callback을 주입하는 범용 확인
modal이고, 13번에서는 그중 LoginPage 경로만 끝까지 추적합니다.

### 전체 실제 코드

```jsx
import { useState } from "react";
import { useNavigate } from "react-router-dom";

import { authApi } from "../../api/authApi.js";
import { authStorage } from "../../auth/authStorage.js";
import ConfirmModal from "../../components/common/ConfirmModal.jsx";
import LoginForm from "../../components/auth/LoginForm.jsx";
import AuthLayout from "../../components/layout/AuthLayout.jsx";
import {
    getErrorMessage,
    hasErrorCode,
} from "../../utils/errorMessage.js";
import { isValidEmail, isValidPassword } from "../../utils/validation.js";
import "../../styles/auth.css";

const EMAIL_EMPTY_MESSAGE = "*이메일을 입력해주세요.";
const EMAIL_INVALID_MESSAGE = "*올바른 이메일 주소 형식을 입력해주세요.";
const PASSWORD_EMPTY_MESSAGE = "*비밀번호를 입력해주세요.";
const PASSWORD_INVALID_MESSAGE = "*비밀번호 형식을 확인해주세요.";

function getFieldError(name, value) {
    if (name === "email") {
        const trimmedEmail = value.trim();

        if (!trimmedEmail) {
            return EMAIL_EMPTY_MESSAGE;
        }
        if (!isValidEmail(trimmedEmail)) {
            return EMAIL_INVALID_MESSAGE;
        }
    }

    if (name === "password") {
        if (!value) {
            return PASSWORD_EMPTY_MESSAGE;
        }
        if (!isValidPassword(value)) {
            return PASSWORD_INVALID_MESSAGE;
        }
    }

    return "";
}

function LoginPage() {
    const navigate = useNavigate();
    const [form, setForm] = useState({ email: "", password: "" });
    const [errors, setErrors] = useState({ email: "", password: "" });
    const [isSubmitting, setIsSubmitting] = useState(false);
    const [isSuspendedModalOpen, setIsSuspendedModalOpen] = useState(false);
    const [suspendedMessage, setSuspendedMessage] = useState("");
    const isValid = isValidEmail(form.email.trim()) && isValidPassword(form.password);

    function handleChange(event) {
        const { name, value } = event.target;
        setForm((previous) => ({ ...previous, [name]: value }));
        setErrors((previous) => ({
            ...previous,
            [name]: getFieldError(name, value),
        }));
    }

    async function handleSubmit(event) {
        event.preventDefault();

        if (!isValid) {
            setErrors({
                email: getFieldError("email", form.email),
                password: getFieldError("password", form.password),
            });
            return;
        }

        try {
            setIsSubmitting(true);
            setErrors({ email: "", password: "" });
            const result = await authApi.login({
                email: form.email.trim(),
                password: form.password,
            });
            authStorage.setAccessToken(result.accessToken);
            navigate("/posts", { replace: true });
        } catch (error) {
            if (hasErrorCode(error, "SUSPENDED_ACCOUNT")) {
                setSuspendedMessage(getErrorMessage(error));
                setIsSuspendedModalOpen(true);
                return;
            }
            setErrors((previous) => ({
                ...previous,
                password: `*${getErrorMessage(
                    error,
                    "이메일 또는 비밀번호를 확인해주세요.",
                )}`,
            }));
        } finally {
            setIsSubmitting(false);
        }
    }

    return (
        <AuthLayout
            pageClassName="login-page"
```

`useState`는 `[현재값, setter]` 배열을 반환하는 React Hook입니다. setter를 호출하면
컴포넌트가 다시 렌더링되고, 새 `form` 값이 LoginForm의 controlled input에 전달됩니다.

### 코드 일부와 바로 아래 설명: 입력 event와 computed property

```jsx
function handleChange(event) {
    const { name, value } = event.target; // 실제 input의 name/value를 구조분해한다.
    setForm((previous) => ({ ...previous, [name]: value })); // 기존 field는 보존하고 해당 이름만 동적으로 갱신한다.
    setErrors((previous) => ({
        ...previous,
        [name]: getFieldError(name, value), // 입력한 field만 즉시 재검증한다.
    }));
}
```

`[name]`은 computed property name입니다. `name`이 `email`이면 `{ email: value }`가 되고,
`password`이면 `{ password: value }`가 됩니다. `...previous`를 먼저 펼쳐야 다른 입력
field가 사라지지 않습니다. 함수형 setter의 `previous`는 React가 직전 state를 넣어 주는
값이라 빠른 연속 입력에도 최신 값을 기준으로 갱신할 수 있습니다.

### 코드 일부와 바로 아래 설명: submit 정상·실패 흐름

```jsx
async function handleSubmit(event) {
    event.preventDefault(); // 브라우저의 페이지 reload를 막는다.

    if (!isValid) {
        setErrors({ // 두 field의 현재 값을 모두 검사해 문구를 채운다.
            email: getFieldError("email", form.email),
            password: getFieldError("password", form.password),
        });
        return; // API를 호출하지 않고 현재 함수만 끝낸다.
    }

    try {
        setIsSubmitting(true); // 버튼을 disabled로 만든다.
        setErrors({ email: "", password: "" }); // 이전 오류를 지운다.
        const result = await authApi.login({ // authApi→api.request→fetch 결과를 기다린다.
            email: form.email.trim(), // 양 끝 공백만 제거해서 backend에 보낸다.
            password: form.password,
        });
        authStorage.setAccessToken(result.accessToken); // 응답 body token을 localStorage에 저장한다.
        navigate("/posts", { replace: true }); // 성공 후 게시글 route로 이동한다.
    } catch (error) {
        if (hasErrorCode(error, "SUSPENDED_ACCOUNT")) { // 정지 계정은 field가 아닌 modal 흐름이다.
            setSuspendedMessage(getErrorMessage(error));
            setIsSuspendedModalOpen(true);
            return;
        }
        setErrors((previous) => ({ // 그 외 로그인 실패는 password 아래에 표시한다.
            ...previous,
            password: `*${getErrorMessage(error, "이메일 또는 비밀번호를 확인해주세요.")}`,
        }));
    } finally {
        setIsSubmitting(false); // 성공·실패·return 여부와 관계없이 submitting을 해제한다.
    }
}
```

`try/catch/finally`는 Promise rejection을 화면 상태로 바꾸는 경계입니다. `await`한
`authApi.login()`이 성공하면 token 저장과 route 이동으로 이어지고, Error가 reject되면
`catch`에서 code별 UI를 선택합니다. `finally`는 catch 안에서 `return`해도 실행되므로
정지 계정 modal을 연 뒤에도 버튼이 계속 “가입 중/로그인 중”으로 남지 않습니다.

## 13.15 로그인 실행 흐름 한 번에 연결하기

1. 사용자가 `/login` 화면에서 email/password를 입력합니다.
2. `LoginPage.handleChange()`가 `event.target.name`으로 `form.email` 또는 `form.password`를
   갱신하고, `getFieldError()` 결과를 같은 이름의 `errors` key에 저장합니다.
3. `LoginForm`은 page가 준 `form`, `errors`, `isValid`를 controlled input과 button에
   전달합니다. 버튼은 유효하지 않거나 제출 중이면 disabled입니다.
4. submit event가 `LoginPage.handleSubmit()`으로 올라옵니다. `preventDefault()`가
   브라우저 기본 reload를 막고, `isValid`가 false이면 오류만 채운 뒤 API를 호출하지
   않습니다.
5. 유효하면 `authApi.login()`이 `POST /sessions`와 JSON body를 만들고 `api.request()`를
   호출합니다.
6. `api.js`의 `sendRequest()`가 access token(있다면), JSON header, Cookie credentials를
   포함해 `fetch`를 실행합니다. 로그인 endpoint는 `NO_REFRESH_REQUESTS`에 있으므로
   401이 와도 로그인 요청이 다시 refresh를 부르는 loop에 들어가지 않습니다.
7. 응답이 성공하면 `request()`가 JSON data를 반환하고, LoginPage가
   `result.accessToken`을 `authStorage.setAccessToken()`으로 저장한 다음
   `navigate("/posts", { replace: true })`를 실행합니다.
8. 응답이 실패하면 `createRequestError()`가 `Error.code`, `Error.status`, `Error.data`를
   만든 뒤 catch로 전달합니다. `SUSPENDED_ACCOUNT`면 `ConfirmModal`을 열고, 그 외에는
   `getErrorMessage()`를 password 오류 문구로 표시합니다.
9. `finally`가 `isSubmitting`을 false로 바꾸고, modal 확인 시 부모 state가 modal을 닫고
   message를 비웁니다.

이 흐름에서 `LoginForm`은 화면 event 연결만 담당하고, `LoginPage`는 업무 순서와 상태를
담당하며, `authApi`는 endpoint 계약을 담당하고, `api.js`는 전역 HTTP·인증 정책을 담당합니다.

2. 입력 event는 `SignupPage.handleChange()`로 올라가 `form`의 동적 key를 바꾸고, 기존
   해당 오류를 지웁니다. blur event는 `validateField()`를 실행해 해당 field의 오류를
   다시 표시합니다.
3. 이미지 label을 클릭하면 연결된 file input이 열립니다. `ProfileImagePicker`는
   `onImageChange`를 SignupPage에 전달합니다.
4. `handleImageChange()`가 첫 File의 type·size를 `getImageFileError()`로 검사합니다.
   실패하면 input·File·preview를 지우고 `alert` 후 끝납니다. 성공하면 원본 File을
   state에 저장하고 `fileToDataUrl()`의 Promise가 끝난 Data URL을 `previewUrl`에 저장합니다.
5. `SignupForm`은 네 field와 이미지 picker에 state·callback을 주입합니다. `errors` 초기값은
   빈 object이므로 사용하지 않은 오류 key는 `undefined`이고 화면에 표시되지 않습니다.
6. submit 시 `Object.keys(form)` 전체를 `validateField()`에 통과시켜 `nextErrors`를 만들고,
   `isValid`가 false이면 이 오류만 표시하고 API를 호출하지 않습니다.
7. 유효하면 `authApi.signup()`이 `POST /users` body를 만듭니다. `...form`에 trimmed
   email/nickname과 Data URL 또는 null인 profileImage를 덮어써 전달합니다.
8. `api.js`가 fetch를 실행합니다. 회원가입 endpoint도 `NO_REFRESH_REQUESTS`에 있으므로
   로그인 전 401이 refresh loop로 연결되지 않습니다.
9. 성공 response가 반환되면 `/login`으로 이동합니다. 오류면 Error code가
   `EXISTED_EMAIL`, `EXISTED_NICKNAME`인지 확인해 각각 해당 field에 문구를 넣고, 그 외는
   email field에 일반 문구를 넣습니다. 마지막으로 `finally`가 제출 상태를 해제합니다.

회원가입에서 client validation과 backend validation은 중복처럼 보여도 책임이 다릅니다.
client 검사는 입력 중 빠른 피드백을 주고, backend 검사는 조작된 요청·중복 데이터·보안
규칙을 최종적으로 차단합니다. 이 문서에서는 같은 정규식 문법을 게시글 form에서 다시
설명하지 않고, 새로 추가되는 field와 호출 위치만 확인합니다.

## 13.19 모범 답안과 오답 정리

### 1번

- 모범 답안: `authApi`는 endpoint·method·payload를 업무 이름으로 묶고, 실제 fetch·token·401
  refresh·Error 변환은 모든 API가 공유해야 하므로 `api.request()`에 위임합니다.
- 오답 포인트: authApi가 직접 fetch한다고 이해하면 공통 인증 정책이 어디서 적용되는지 놓칩니다.

### 2번

- 모범 답안: 여러 API 요청이 동시에 401을 받아도 첫 refresh의 Promise 하나를 함께 기다리게
  해 refresh HTTP 요청을 하나로 줄이고, 완료 후 `finally`에서 null로 되돌려 다음 만료에
  다시 요청할 수 있게 합니다.
- 오답 포인트: 매 401마다 refresh를 새로 만들면 동시 요청 수만큼 rotate 요청이 생깁니다.

### 3번

- 모범 답안: JavaScript object는 존재하지 않는 property를 읽으면 오류 대신 `undefined`를
  반환하고, React는 `undefined`를 텍스트로 렌더링하지 않습니다. 사용자가 blur·submit하거나
  backend 오류가 발생하면 해당 key가 `setErrors`로 추가됩니다.
- 오답 포인트: `errors`가 처음부터 email/password/passwordCheck/nickname을 모두 선언한
  별도 class라고 생각하면 실제 초기값 `{}`와 다릅니다.

### 4번

- 모범 답안: label의 `htmlFor`는 input의 `id`를 가리켜 label 클릭이 해당 file picker를
  엽니다. 값이 다르면 label을 눌러도 연결된 input이 열리지 않습니다.
- 오답 포인트: `accept`가 label과 input을 연결한다고 생각하면 안 됩니다. `accept`는 파일
  형식 선택 힌트입니다.

### 5번

- 모범 답안: FileReader의 `onload`·`onerror`는 나중에 실행되는 비동기 event이므로 Promise로
  감싸야 page가 `await`로 Data URL 생성 완료를 기다리고, 실패를 `catch`로 받을 수 있습니다.
- 오답 포인트: `readAsDataURL()` 호출 직후 반환값이 Data URL 문자열이라고 생각하면 안 됩니다.

### 6번

- 모범 답안: refresh endpoint 자체가 401이면 일반 request가 다시 refresh를 호출하는 재귀적
  loop가 생길 수 있습니다. 제외 목록은 로그인·refresh·로그아웃·회원가입 요청의 401을
  재발급 경로에서 제외합니다.
- 오답 포인트: 제외 목록이 401을 성공으로 만든다고 생각하면 안 됩니다. 해당 응답은 여전히
  오류로 page에 전달됩니다.

### 7번

- 모범 답안: 정지 계정은 단순히 password 한 field가 틀린 것이 아니라 계정 사용 제한이라는
  별도 안내가 필요하므로 `suspendedMessage`와 `isSuspendedModalOpen` state로 modal에
  표시합니다.
- 오답 포인트: 모든 backend 오류를 동일한 password 문구로 처리한다고 보면 현재 code별
  분기를 놓칩니다.
